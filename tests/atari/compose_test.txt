; Each run selects one host/path/leaf explicitly, primes every result byte to
; $ff, then requires the harness to report those inputs, 64 completed
; capacity checks and success. The last case also tests the 196-byte boundary.
; Separate runs keep each group below soft65c02's instruction limit.
memory load atari "${BINARY_PATH}"
symbols load "${SYMBOLS_PATH}"
registers flush

; host 0, path 0, leaf 0
memory write $_test_host 0x(00)
memory write $_test_path 0x(00)
memory write $_test_leaf 0x(00)
memory write $_test_result 0x(FF,FF,FF,FF,FF,FF)
run $init_harness until CP = $end_init_harness
assert $_test_result = 0x00 $$host 0, path 0, leaf 0: comparison succeeded$$
assert $_test_result + 1 = 0x00 $$host 0 executed$$
assert $_test_result + 2 = 0x00 $$path 0 executed$$
assert $_test_result + 3 = 0x00 $$leaf 0 executed$$
assert $_test_result + 4 = 0x40 $$64 capacities completed$$
